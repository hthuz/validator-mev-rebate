```go
// Builder评分数据结构
type BuilderScore struct {
    BuilderID       string    // Builder唯一标识，如"flashbots"
    TotalReward     float64   // 累计收益（单位：ETH）
    SelectCount     int64     // 被选择次数
    MeanReward      float64   // 平均收益 μᵢ = TotalReward / SelectCount
    RewardVariance  float64   // 收益方差
    LastUpdateTime  int64     // 最后更新时间戳（Unix时间）
    LastUpdateBlock uint64    // 最后更新的区块号

    // 多维度指标
    Metrics struct {
        AvgLatency      float64 // 平均响应延迟（毫秒）
        SuccessRate     float64 // 成功率（0-1）
        TimeoutCount    int64   // 超时次数
        TotalAttempts   int64   // 总尝试次数
        ComplianceScore float64 // 合规性评分（0-1）
    }

    // 异常检测
    AnomalyFlags struct {
        BidPaymentGap   float64 // 出价与实付差异累计
        LowInclusionRate bool   // 低包含率标记
        Blacklisted     bool    // 黑名单标记
    }
}

// 全局Builder池
type BuilderPool struct {
    Builders      map[string]*BuilderScore // BuilderID -> Score
    TotalSelections int64                   // 全局选择总次数 N
    CurrentBlock    uint64                  // 当前区块高度
    mutex           sync.RWMutex            // 并发控制
}
```

### 1.3 核心算法实现

#### 1.3.1 UCB评分计算

```go
// UCB评分计算函数
func (bp *BuilderPool) CalculateUCB(builderID string, explorationCoeff float64) (float64, error) {
    bp.mutex.RLock()
    defer bp.mutex.RUnlock()

    builder, exists := bp.Builders[builderID]
    if !exists {
        return 0, fmt.Errorf("builder %s not found", builderID)
    }

    // 如果从未被选择，返回无穷大（确保被探索）
    if builder.SelectCount == 0 {
        return math.Inf(1), nil
    }

    // 计算探索奖励项
    // explorationBonus = c × √(2 × ln(N) / nᵢ)
    N := float64(bp.TotalSelections)
    ni := float64(builder.SelectCount)

    explorationBonus := explorationCoeff * math.Sqrt(2 * math.Log(N) / ni)

    // UCB = μᵢ + c × √(2 × ln(N) / nᵢ)
    ucbScore := builder.MeanReward + explorationBonus

    return ucbScore, nil
}

// 多维度UCB评分计算（扩展版本）
func (bp *BuilderPool) CalculateMultiDimensionalUCB(
    builderID string,
    explorationCoeff float64,
    weights map[string]float64, // 各维度权重
) (float64, error) {
    bp.mutex.RLock()
    defer bp.mutex.RUnlock()

    builder, exists := bp.Builders[builderID]
    if !exists {
        return 0, fmt.Errorf("builder %s not found", builderID)
    }

    if builder.SelectCount == 0 {
        return math.Inf(1), nil
    }

    // 归一化各维度指标到 [0, 1]
    rewardScore := normalizeReward(builder.MeanReward, 0.0, 1.0)
    latencyScore := normalizeLatency(builder.Metrics.AvgLatency, 0, 500) // 0-500ms
    successScore := builder.Metrics.SuccessRate
    complianceScore := builder.Metrics.ComplianceScore

    // 加权综合评分
    baseScore := weights["reward"] * rewardScore +
                weights["latency"] * latencyScore +
                weights["success"] * successScore +
                weights["compliance"] * complianceScore

    // 探索奖励
    N := float64(bp.TotalSelections)
    ni := float64(builder.SelectCount)
    explorationBonus := explorationCoeff * math.Sqrt(2 * math.Log(N) / ni)

    // 异常惩罚
    anomalyPenalty := 0.0
    if builder.AnomalyFlags.Blacklisted {
        return -math.Inf(1), nil // 黑名单直接排除
    }
    if builder.AnomalyFlags.LowInclusionRate {
        anomalyPenalty = 0.1 // 降低10%
    }
    if builder.AnomalyFlags.BidPaymentGap > 0.1 {
        anomalyPenalty += builder.AnomalyFlags.BidPaymentGap * 0.5
    }

    ucbScore := baseScore + explorationBonus - anomalyPenalty

    return ucbScore, nil
}

// 辅助函数：收益归一化
func normalizeReward(reward, min, max float64) float64 {
    if max-min == 0 {
        return 0.5
    }
    normalized := (reward - min) / (max - min)
    return math.Max(0, math.Min(1, normalized))
}

// 辅助函数：延迟归一化（越小越好）
func normalizeLatency(latency, min, max float64) float64 {
    if max-min == 0 {
        return 0.5
    }
    normalized := 1.0 - (latency-min)/(max-min)
    return math.Max(0, math.Min(1, normalized))
}
```

#### 1.3.2 Builder选择决策

```go
// Builder选择主函数
func (bp *BuilderPool) SelectBuilder(
    bundle *MevBundle,
    explorationCoeff float64,
    weights map[string]float64,
) (*SelectionDecision, error) {
    bp.mutex.RLock()
    defer bp.mutex.RUnlock()

    if len(bp.Builders) == 0 {
        return nil, fmt.Errorf("no builders available")
    }

    bestBuilder := ""
    bestScore := -math.Inf(1)
    allScores := make(map[string]float64)

    // 计算所有Builder的UCB分数
    for builderID := range bp.Builders {
        score, err := bp.CalculateMultiDimensionalUCB(builderID, explorationCoeff, weights)
        if err != nil {
            continue
        }

        allScores[builderID] = score

        if score > bestScore {
            bestScore = score
            bestBuilder = builderID
        }
    }

    // 判断选择原因（探索 vs 利用）
    builder := bp.Builders[bestBuilder]
    reason := "exploitation"

    // 如果该Builder被选择次数显著少于平均值，标记为探索
    avgSelections := float64(bp.TotalSelections) / float64(len(bp.Builders))
    if float64(builder.SelectCount) < avgSelections * 0.5 {
        reason = "exploration"
    }

    decision := &SelectionDecision{
        BundleHash:      bundle.BundleHash,
        SelectedBuilder: bestBuilder,
        UCBScore:        bestScore,
        SelectionReason: reason,
        Timestamp:       time.Now().Unix(),
    }

    return decision, nil
}
```

#### 1.3.3 评分更新

```go
// 更新Builder评分（收到实际收益后）
func (bp *BuilderPool) UpdateBuilderScore(
    builderID string,
    actualReward float64,
    latency float64,
    success bool,
    blockNumber uint64,
) error {
    bp.mutex.Lock()
    defer bp.mutex.Unlock()

    builder, exists := bp.Builders[builderID]
    if !exists {
        // 新Builder初始化
        builder = &BuilderScore{
            BuilderID: builderID,
        }
        bp.Builders[builderID] = builder
    }

    // 更新累计数据
    builder.TotalReward += actualReward
    builder.SelectCount++
    bp.TotalSelections++

    // 更新平均收益
    builder.MeanReward = builder.TotalReward / float64(builder.SelectCount)

    // 更新方差（增量计算）
    if builder.SelectCount > 1 {
        oldVariance := builder.RewardVariance
        n := float64(builder.SelectCount)
        delta := actualReward - builder.MeanReward
        builder.RewardVariance = ((n-1)*oldVariance + delta*delta) / n
    }

    // 更新延迟指标（指数移动平均）
    alpha := 0.1 // 平滑系数
    if builder.Metrics.AvgLatency == 0 {
        builder.Metrics.AvgLatency = latency
    } else {
        builder.Metrics.AvgLatency = alpha*latency + (1-alpha)*builder.Metrics.AvgLatency
    }

    // 更新成功率
    builder.Metrics.TotalAttempts++
    if success {
        // 增量更新成功率
        successCount := builder.Metrics.SuccessRate * float64(builder.Metrics.TotalAttempts-1)
        if success {
            successCount++
        }
        builder.Metrics.SuccessRate = successCount / float64(builder.Metrics.TotalAttempts)
    } else {
        builder.Metrics.TimeoutCount++
        successCount := builder.Metrics.SuccessRate * float64(builder.Metrics.TotalAttempts-1)
        builder.Metrics.SuccessRate = successCount / float64(builder.Metrics.TotalAttempts)
    }

    // 更新时间戳
    builder.LastUpdateTime = time.Now().Unix()
    builder.LastUpdateBlock = blockNumber

    return nil
}
```

#### 1.3.4 自适应参数调整

```go
// 自适应调整探索系数
func (ac *AdaptiveController) AdjustExplorationCoeff(
    recentRewards []float64, // 最近N个区块的收益序列
) float64 {
    if len(recentRewards) < ac.Params.VolatilityWindow {
        return ac.Params.BaselineCoeff
    }

    // 计算市场波动率（标准差）
    mean := calculateMean(recentRewards)
    variance := 0.0
    for _, r := range recentRewards {
        variance += math.Pow(r-mean, 2)
    }
    variance /= float64(len(recentRewards))
    volatility := math.Sqrt(variance)

    ac.Params.MarketVolatility = volatility

    oldCoeff := ac.Params.ExplorationCoeff
    newCoeff := oldCoeff
    reason := ""

    // 高波动：降低探索系数，专注利用
    if volatility > ac.Params.HighVolatilityThreshold {
        newCoeff = ac.Params.MinCoeff
        reason = "high_volatility"
    }
    // 低波动：提高探索系数，增加探索
    else if volatility < ac.Params.LowVolatilityThreshold {
        newCoeff = ac.Params.MaxCoeff
        reason = "low_volatility"
    }
    // 中等波动：使用基准系数
    else {
        newCoeff = ac.Params.BaselineCoeff
        reason = "normal_volatility"
    }

    // 平滑调整（避免剧烈变化）
    smoothingFactor := 0.3
    newCoeff = smoothingFactor*newCoeff + (1-smoothingFactor)*oldCoeff

    // 限制范围
    newCoeff = math.Max(ac.Params.MinCoeff, math.Min(ac.Params.MaxCoeff, newCoeff))

    // 记录调整
    if newCoeff != oldCoeff {
        record := AdjustmentRecord{
            BlockNumber: ac.CurrentBlock,
            OldCoeff:    oldCoeff,
            NewCoeff:    newCoeff,
            Reason:      reason,
        }
        ac.Params.AdjustmentHistory = append(ac.Params.AdjustmentHistory, record)
    }

    ac.Params.ExplorationCoeff = newCoeff
    return newCoeff
}

func calculateMean(values []float64) float64 {
    sum := 0.0
    for _, v := range values {
        sum += v
    }
    return sum / float64(len(values))
}
```

```go
// 贝叶斯融合：结合本地经验和全局知识
func (hs *HybridBuilderScore) FuseScores(beta float64) float64 {
    // α = n_local / (n_local + β)
    // β是平滑参数，控制信任全局数据的程度

    if hs.LocalCount == 0 {
        // 纯新Builder，完全依赖全局数据
        hs.FusedMean = hs.GlobalMean
        hs.FusionWeight = 0.0
        return hs.FusedMean
    }

    alpha := float64(hs.LocalCount) / (float64(hs.LocalCount) + beta)
    hs.FusionWeight = alpha

    // 融合评分 = α × μ_local + (1-α) × μ_global
    hs.FusedMean = alpha*hs.LocalMean + (1-alpha)*hs.GlobalMean

    return hs.FusedMean
}

// UCB计算（使用融合评分）
func (hs *HybridBuilderScore) CalculateHybridUCB(
    totalLocalSelections int64,
    explorationCoeff float64,
    beta float64,
) float64 {
    fusedMean := hs.FuseScores(beta)

    if hs.LocalCount == 0 {
        return math.Inf(1) // 未本地测试，优先探索
    }

    // 探索奖励基于本地选择次数（鼓励本地验证）
    N := float64(totalLocalSelections)
    ni := float64(hs.LocalCount)

    explorationBonus := explorationCoeff * math.Sqrt(2*math.Log(N)/ni)

    // 融合置信度奖励
    // 如果全局数据量大，置信度高，额外加分
    confidenceBonus := 0.0
    if hs.GlobalCount > 100 {
        confidenceBonus = 0.01 * math.Log(float64(hs.GlobalCount))
    }

    ucb := fusedMean + explorationBonus + confidenceBonus

    return ucb
}
```

```go
// 从全局池拉取最新评分
func (v *Validator) PullFromGlobal(builderID string) (*GlobalBuilderScore, error) {
    globalScore, err := v.GlobalDB.GetGlobalScore(builderID)
    if err != nil {
        return nil, err
    }

    // 重新计算全局聚合评分
    totalWeight := 0.0
    weightedSum := 0.0

    for _, contrib := range globalScore.Contributors {
        weight := float64(contrib.LocalCount)
        totalWeight += weight
        weightedSum += contrib.LocalMean * weight
    }

    if totalWeight > 0 {
        globalScore.GlobalMean = weightedSum / totalWeight
        globalScore.GlobalCount = int64(totalWeight)
    }

    return globalScore, nil
}
```

```go
type MarketState string

const (
    StateNormal      MarketState = "normal"       // 正常状态
    StateHighVolatile MarketState = "high_volatile" // 高波动
    StateLowVolatile  MarketState = "low_volatile"  // 低波动
    StateAnomaly     MarketState = "anomaly"       // 异常事件

// 市场监控器
type MarketMonitor struct {
    RewardHistory   []float64      // 最近N个区块的收益
    WindowSize      int            // 监控窗口大小
    CurrentState    MarketState    // 当前市场状态

    // 阈值配置
    NormalMean      float64        // 正常期望收益
    NormalStdDev    float64        // 正常标准差
    HighVolThreshold float64       // 高波动阈值
    LowVolThreshold  float64       // 低波动阈值
    AnomalyMultiple  float64       // 异常倍数（如3倍标准差）
}

// 实时检测市场状态
func (mm *MarketMonitor) DetectMarketState(newReward float64) MarketState {
    // 添加新数据
    mm.RewardHistory = append(mm.RewardHistory, newReward)
    if len(mm.RewardHistory) > mm.WindowSize {
        mm.RewardHistory = mm.RewardHistory[1:] // 移除最旧的
    }

    if len(mm.RewardHistory) < mm.WindowSize {
        return StateNormal // 数据不足，默认正常
    }

    // 计算当前统计量
    mean := calculateMean(mm.RewardHistory)
    stdDev := calculateStdDev(mm.RewardHistory, mean)

    // 检测异常（突发事件）
    if newReward > mm.NormalMean+mm.AnomalyMultiple*mm.NormalStdDev {
        mm.CurrentState = StateAnomaly
        return StateAnomaly
    }

    // 检测波动率
    if stdDev > mm.HighVolThreshold {
        mm.CurrentState = StateHighVolatile
        return StateHighVolatile
    }

    if stdDev < mm.LowVolThreshold {
        mm.CurrentState = StateLowVolatile
        return StateLowVolatile
    }

    mm.CurrentState = StateNormal
    return StateNormal
}

func calculateStdDev(values []float64, mean float64) float64 {
    variance := 0.0
    for _, v := range values {
        variance += math.Pow(v-mean, 2)
    }
    variance /= float64(len(values))
    return math.Sqrt(variance)
}
```

```go
    controller.UpdateFrequency = config.UpdateFrequency
    controller.TopNSelection = config.TopNSelection

    controller.LogAdjustment(state, oldCoeff, adjustedCoeff)
```

```go
type DetectorFunc func(*BuilderScore, *MevBundle) float64

// 检测1：出价与实付差异
func BidPaymentGapDetector(builder *BuilderScore, bundle *MevBundle) float64 {
    if builder.SelectCount < 3 {
        return 0.0 // 数据不足
    }

    gap := builder.AnomalyFlags.BidPaymentGap
    avgReward := builder.MeanReward

    if avgReward == 0 {
        return 0.0
    }

    // 差异比例
    gapRatio := gap / avgReward

    // 异常分数：差异超过20%开始计分
    if gapRatio > 0.2 {
        return (gapRatio - 0.2) * 5.0 // 每超过10%，+0.5分
    }

    return 0.0
}

// 检测2：低包含率
func LowInclusionRateDetector(builder *BuilderScore, bundle *MevBundle) float64 {
    totalAttempts := builder.Metrics.TotalAttempts
    if totalAttempts < 10 {
        return 0.0
    }

    successRate := builder.Metrics.SuccessRate

    // 正常包含率应 > 40%
    if successRate < 0.4 {
        return (0.4 - successRate) * 10.0 // 每低10%，+1分
    }

    return 0.0
}

// 检测3：响应时间异常
func LatencyAnomalyDetector(builder *BuilderScore, bundle *MevBundle) float64 {
    avgLatency := builder.Metrics.AvgLatency

    // 正常延迟应 < 200ms
    if avgLatency > 200 {
        return (avgLatency - 200) / 100.0 // 每超100ms，+1分
    }

    return 0.0
}

// 检测4：收益持续低于市场
func UnderperformanceDetector(
    builder *BuilderScore,
    bundle *MevBundle,
    marketAvg float64,
) float64 {
    if builder.SelectCount < 20 {
        return 0.0
    }

    builderMean := builder.MeanReward

    // 持续低于市场平均20%以上
    if builderMean < marketAvg*0.8 {
        return (marketAvg*0.8 - builderMean) / marketAvg * 5.0
    }

    return 0.0
}

// 综合异常评分
func (ad *AnomalyDetector) CalculateAnomalyScore(
    builder *BuilderScore,
    bundle *MevBundle,
    marketAvg float64,
) float64 {
    totalScore := 0.0

    for _, detector := range ad.Detectors {
        score := detector(builder, bundle)
        totalScore += score
    }

    // 添加自定义检测
    totalScore += UnderperformanceDetector(builder, bundle, marketAvg)

    return totalScore
}

    builderID string,

        return "normal" // 正常
```

```go
// 使用读写锁优化并发
type ConcurrentBuilderPool struct {
    builders sync.Map // builderID -> *BuilderScore (lock-free读)

    // 分片计数器（减少锁竞争）
    shardedCounters []*ShardedCounter
    shardCount      int
}

type ShardedCounter struct {
    mu    sync.Mutex
    count int64
}

// 原子操作累加
func (cp *ConcurrentBuilderPool) IncrementTotalSelections() {
    shard := cp.shardedCounters[int(time.Now().UnixNano())%cp.shardCount]
    shard.mu.Lock()
    shard.count++
    shard.mu.Unlock()
}

// 聚合计数
func (cp *ConcurrentBuilderPool) GetTotalSelections() int64 {
    total := int64(0)
    for _, shard := range cp.shardedCounters {
        shard.mu.Lock()
        total += shard.count
        shard.mu.Unlock()
    }
    return total
}
```

### 6.2 增量计算优化

```go
// 避免重复计算UCB，使用缓存
type CachedUCBCalculator struct {
    cache map[string]*UCBCache
    mu    sync.RWMutex
    ttl   time.Duration
}

type UCBCache struct {
    Score      float64
    ComputedAt time.Time
    ValidUntil time.Time
}

func (calc *CachedUCBCalculator) GetUCB(
    builderID string,
    pool *BuilderPool,
    explorationCoeff float64,
) (float64, bool) {
    calc.mu.RLock()
    cached, exists := calc.cache[builderID]
    calc.mu.RUnlock()

    if exists && time.Now().Before(cached.ValidUntil) {
        return cached.Score, true // 缓存命中
    }

    // 缓存未命中，重新计算
    score, _ := pool.CalculateUCB(builderID, explorationCoeff)

    calc.mu.Lock()
    calc.cache[builderID] = &UCBCache{
        Score:      score,
        ComputedAt: time.Now(),
        ValidUntil: time.Now().Add(calc.ttl),
    }
    calc.mu.Unlock()

    return score, false
}

// 批量预计算（在选择前）
func (calc *CachedUCBCalculator) PrecomputeAll(
    pool *BuilderPool,
    explorationCoeff float64,
) {
    for builderID := range pool.Builders {
        calc.GetUCB(builderID, pool, explorationCoeff)
    }
}
```
